    （A）创建 scripts/migrate-data.ts：

    使用更好 SQLite 兼容的库（better-sqlite3 或 sql.js）读取 mwlab.db，然后逐表使用 @supabase/supabase-js 写入 Supabase。

    ```typescript
    import Database from 'better-sqlite3'
    import { createClient } from '@supabase/supabase-js'

    const DB_PATH = process.env.SQLITE_DB_PATH || './mwlab.db'

    const supabase = createClient(
      process.env.NEXT_PUBLIC_SUPABASE_URL!,
      process.env.SUPABASE_SERVICE_ROLE_KEY!
    )

    interface TableConfig {
      name: string
      batchSize: number
      transform?: (row: any) => any
    }

    const TABLES: TableConfig[] = [
      {
        name: 'exhibition_brand',
        batchSize: 100,
        transform: (row: any) => ({
          ...row,
          // datetime → ISO string for Supabase
          created_at: row.created_at ? new Date(row.created_at).toISOString() : new Date().toISOString(),
          updated_at: row.updated_at ? new Date(row.updated_at).toISOString() : new Date().toISOString(),
        }),
      },
      {
        name: 'exhibition_edition',
        batchSize: 100,
        transform: (row: any) => ({
          ...row,
          recorded_at: row.recorded_at ? new Date(row.recorded_at).toISOString() : new Date().toISOString(),
          date_start: row.date_start || null,
          date_end: row.date_end || null,
        }),
      },
      {
        name: 'data_provenance',
        batchSize: 500,
        transform: (row: any) => ({
          ...row,
          crawled_at: row.crawled_at ? new Date(row.crawled_at).toISOString() : new Date().toISOString(),
        }),
      },
      {
        name: 'crawl_log',
        batchSize: 50,
        transform: (row: any) => ({
          ...row,
          started_at: row.started_at ? new Date(row.started_at).toISOString() : null,
          finished_at: row.finished_at ? new Date(row.finished_at).toISOString() : null,
          created_at: row.created_at ? new Date(row.created_at).toISOString() : new Date().toISOString(),
        }),
      },
      {
        name: 'manual_tag_history',
        batchSize: 200,
        transform: (row: any) => ({
          ...row,
          changed_at: row.changed_at ? new Date(row.changed_at).toISOString() : new Date().toISOString(),
        }),
      },
      // user 表不迁移（由 seed-users.ts 替代）
    ]

    async function main() {
      const sqlite = new Database(DB_PATH)

      for (const table of TABLES) {
        console.log(`--- Migrating ${table.name} ---`)
        const rows = sqlite.prepare(`SELECT * FROM ${table.name}`).all() as any[]
        console.log(`  Read ${rows.length} rows from SQLite`)

        let inserted = 0
        for (let i = 0; i < rows.length; i += table.batchSize) {
          const batch = rows.slice(i, i + table.batchSize)
          const transformed = table.transform ? batch.map(table.transform) : batch

          const { error } = await supabase
            .from(table.name)
            .upsert(transformed, { onConflict: undefined })

          if (error) {
            console.error(`  ERROR batch ${i / table.batchSize + 1}: ${error.message}`)
          } else {
            inserted += batch.length
            console.log(`  Inserted ${inserted}/${rows.length} rows`)
          }
        }
        console.log(`  Done: ${inserted}/${rows.length} rows migrated`)
      }

      sqlite.close()
      console.log('Migration complete!')
    }

    main().catch(console.error)
    ```

    **关键注意事项**：
    - 使用 upsert（而非 insert）以支持幂等执行
    - SQLite 中的 TEXT 日期格式需要转换为 ISO 8601 字符串（Supabase TIMESTAMPTZ 需要）
    - 批处理插入（batchSize 100-500 条，避免超过 Supabase 请求大小限制）
    - 使用 SERVICE_ROLE_KEY（绕过 RLS，直接写入）
