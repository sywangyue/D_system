  **A) KPI 聚合验证**
  ```bash
  python3 << 'PYEOF'
  import sqlite3, json
  conn = sqlite3.connect('mwlab.db')
  conn.row_factory = sqlite3.Row

  # 总展览面积
  row = conn.execute("""
    SELECT 
      COALESCE(SUM(area_sqm), 0) as total_area,
      COALESCE(SUM(exhibitors_count), 0) as total_exhibitors,
      COALESCE(SUM(visitors_count), 0) as total_visitors,
      COUNT(DISTINCT organizer) as total_organizers
    FROM exhibition_brand b
    JOIN exhibition_edition e ON e.brand_id = b.brand_id
  """).fetchone()
  print(f"  KPI: area={row['total_area']}, exhibitors={row['total_exhibitors']}, visitors={row['total_visitors']}, organizers={row['total_organizers']}")

  # 行业分布
  dist = conn.execute("""
    SELECT industry_l2, COUNT(*) as cnt
    FROM exhibition_brand WHERE industry_l2 != ''
    GROUP BY industry_l2 ORDER BY cnt DESC LIMIT 10
  """).fetchall()
  print(f"  Industry distribution: {len(dist)} categories")
  for d in dist:
      print(f"    {d['industry_l2']}: {d['cnt']}")

  # 城市分布（地图）
  cities = conn.execute("""
    SELECT e.city, COUNT(DISTINCT b.brand_id) as cnt
    FROM exhibition_brand b
    JOIN exhibition_edition e ON e.brand_id = b.brand_id
    WHERE e.city != ''
    GROUP BY e.city ORDER BY cnt DESC LIMIT 10
  """).fetchall()
  print(f"  Top cities: {len(cities)} cities")
  for c in cities:
      print(f"    {c['city']}: {c['cnt']} exhibitions")

  conn.close()
  PYEOF
  ```
