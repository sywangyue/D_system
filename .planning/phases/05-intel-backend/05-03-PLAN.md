```python
#!/usr/bin/env python3
"""
tools/intel/report_writer.py — 调研报告持久化工具

功能：将 Markdown 报告写入文件系统和 intel_report 表。
供 Skill 文件在完成分析后调用（通过 !`bash` 命令或 Skill 内嵌 Python 块）。

用法（命令行）：
  python3 tools/intel/report_writer.py \\
    --type brand_research \\
    --brand_id EXPO-0001 \\
    --content "$(cat /tmp/report_draft.md)"

  python3 tools/intel/report_writer.py \\
    --type industry_research \\
    --industry_l1 "机械和设备" \\
    --content "# 报告内容..."

用法（Python 导入）：
  from tools.intel.report_writer import write_report
  report_id = write_report("brand_research", {"brand_id": "EXPO-0001"}, content_md, brand_id="EXPO-0001")
"""
from __future__ import annotations

import argparse
import json
import sqlite3
from datetime import datetime
from pathlib import Path
from typing import Any

_REPO_ROOT = Path(__file__).resolve().parent.parent.parent
DB_PATH = _REPO_ROOT / "mwlab.db"
REPORTS_DIR = _REPO_ROOT / "reports"

VALID_REPORT_TYPES = frozenset({
    "industry_research",
    "brand_research",
    "batch_prospect",
    "single_prospect",
})


def write_report(
    report_type: str,
    params: dict[str, Any],
    content_md: str,
    brand_id: str | None = None,
    industry_l1: str | None = None,
    industry_l2: str | None = None,
    target_company: str | None = None,
    created_by: str = "claude-code",
) -> int:
    """
    将调研报告写入 Markdown 文件 + intel_report 表。

    返回值: intel_report.id（整数）

    注意：content_md 使用参数化查询写入，支持任意特殊字符（per Pitfall 4）。
    """
    if report_type not in VALID_REPORT_TYPES:
        raise ValueError(
            f"非法 report_type: '{report_type}'。"
            f"允许值: {sorted(VALID_REPORT_TYPES)}"
        )

    # 文件保存
    subdir = REPORTS_DIR / report_type
    subdir.mkdir(parents=True, exist_ok=True)

    date_str = datetime.now().strftime("%Y%m%d-%H%M%S")
    name_part = brand_id or industry_l1 or target_company or "unknown"
    # 替换路径非法字符
    name_part = name_part.replace("/", "_").replace(" ", "_")
    file_name = f"{date_str}-{name_part}.md"
    file_path = subdir / file_name
    file_path.write_text(content_md, encoding="utf-8")

    # 相对路径（用于 DB 记录，相对项目根）
    relative_path = str(file_path.relative_to(_REPO_ROOT))

    conn = sqlite3.connect(str(DB_PATH))
    try:
        cur = conn.execute(
            "INSERT INTO intel_report "
            "(report_type, brand_id, industry_l1, industry_l2, target_company, "
            " params_json, report_md, report_file, created_by) "
            "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
            (
                report_type,
                brand_id,
                industry_l1,
                industry_l2,
                target_company,
                json.dumps(params, ensure_ascii=False),
                content_md,    # 参数化，支持单引号等特殊字符（per Pitfall 4）
                relative_path,
                created_by,
            )
        )
        conn.commit()
        report_id = cur.lastrowid
    finally:
        conn.close()

    print(f"[report_writer] 报告已保存: {relative_path} (report_id={report_id})")
    return report_id


def main() -> None:
    parser = argparse.ArgumentParser(description="调研报告写入工具")
    parser.add_argument(
        "--type", dest="report_type", required=True,
        choices=list(VALID_REPORT_TYPES),
        help="报告类型"
    )
    parser.add_argument("--brand_id", help="品牌 ID（brand_research/batch_prospect 用）")
    parser.add_argument("--industry_l1", help="行业 L1（industry_research 用）")
    parser.add_argument("--industry_l2", help="行业 L2（可选）")
    parser.add_argument("--target_company", help="目标公司名（single_prospect 用）")
    parser.add_argument("--content", required=True, help="Markdown 报告正文")
    parser.add_argument("--created_by", default="claude-code", help="创建者标识")
    args = parser.parse_args()

    params: dict[str, Any] = {}
    if args.brand_id:
        params["brand_id"] = args.brand_id
    if args.industry_l1:
        params["industry_l1"] = args.industry_l1
    if args.industry_l2:
        params["industry_l2"] = args.industry_l2
    if args.target_company:
        params["target_company"] = args.target_company

    report_id = write_report(
        report_type=args.report_type,
        params=params,
        content_md=args.content,
        brand_id=args.brand_id,
        industry_l1=args.industry_l1,
        industry_l2=args.industry_l2,
        target_company=args.target_company,
        created_by=args.created_by,
    )
    print(f"report_id={report_id}")


if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""
tools/intel/export_prospects.py — customer_prospect 批量导出

用法:
  # 导出某次批量挖掘报告的所有 prospect
  python3 tools/intel/export_prospects.py --report_id 42 -o exports/prospects_42.xlsx

  # 导出某展会品牌的所有 prospect
  python3 tools/intel/export_prospects.py --brand_id EXPO-0001 -o exports/expo0001_prospects.xlsx

  # 导出全部 prospect（不过滤）
  python3 tools/intel/export_prospects.py -o exports/all_prospects.xlsx

  # 导出为 CSV
  python3 tools/intel/export_prospects.py --brand_id EXPO-0001 --format csv -o exports/out.csv

依赖: openpyxl（已安装），pandas（已安装）
"""
from __future__ import annotations

import argparse
import csv
import sqlite3
import sys
from datetime import date
from pathlib import Path

_REPO_ROOT = Path(__file__).resolve().parent.parent.parent
if str(_REPO_ROOT) not in sys.path:
    sys.path.insert(0, str(_REPO_ROOT))

from openpyxl import Workbook
from openpyxl.styles import Font

DB_PATH = _REPO_ROOT / "mwlab.db"

# 导出列定义（DB 字段名 → Excel 显示名）
EXPORT_COLUMNS: list[tuple[str, str]] = [
    ("id",               "ID"),
    ("intel_report_id",  "报告ID"),
    ("brand_id",         "展会品牌ID"),
    ("source_type",      "数据来源"),
    ("company_name",     "企业名称"),
    ("qcc_key_no",       "企查查KeyNo"),
    ("credit_code",      "统一社会信用代码"),
    ("oper_name",        "法定代表人"),
    ("start_date",       "成立日期"),
    ("company_status",   "企业状态"),
    ("reg_no",           "注册号"),
    ("address",          "注册地址"),
    ("prospect_score",   "BD评分(1-5)"),
    ("contact_status",   "接触状态"),
    ("notes",            "备注"),
    ("created_at",       "创建时间"),
]

_DB_COLS = [col for col, _ in EXPORT_COLUMNS]
_HEADER_COLS = [label for _, label in EXPORT_COLUMNS]


def _fetch_prospects(
    conn: sqlite3.Connection,
    intel_report_id: int | None,
    brand_id: str | None,
) -> list[sqlite3.Row]:
    conn.row_factory = sqlite3.Row
    sql = "SELECT " + ", ".join(_DB_COLS) + " FROM customer_prospect WHERE 1=1"
    params: list = []
    if intel_report_id is not None:
        sql += " AND intel_report_id = ?"
        params.append(intel_report_id)
    if brand_id:
        sql += " AND brand_id = ?"
        params.append(brand_id)
    sql += " ORDER BY id"
    return list(conn.execute(sql, params).fetchall())


def export_to_excel(
    output_path: Path,
    intel_report_id: int | None = None,
    brand_id: str | None = None,
) -> int:
    """
    导出 customer_prospect 为 Excel。
    返回导出行数（不含表头）。
    """
    conn = sqlite3.connect(str(DB_PATH))
    rows = _fetch_prospects(conn, intel_report_id, brand_id)
    conn.close()

    wb = Workbook()
    ws = wb.active
    ws.title = "Prospects"

    # 表头（加粗，参考 export_for_tagging.py 风格）
    ws.append(_HEADER_COLS)
    for cell in ws[1]:
        cell.font = Font(bold=True)

    # 数据行
    for row in rows:
        ws.append([row[col] for col in _DB_COLS])

    # 列宽自适应（简单实现）
    for i, col_name in enumerate(_HEADER_COLS, start=1):
        ws.column_dimensions[ws.cell(row=1, column=i).column_letter].width = max(
            len(col_name) * 2, 12
        )

    output_path.parent.mkdir(parents=True, exist_ok=True)
    wb.save(str(output_path))
    print(f"[export_prospects] 已导出 {len(rows)} 条记录到: {output_path}")
    return len(rows)


def export_to_csv(
    output_path: Path,
    intel_report_id: int | None = None,
    brand_id: str | None = None,
) -> int:
    """导出 customer_prospect 为 CSV。"""
    conn = sqlite3.connect(str(DB_PATH))
    rows = _fetch_prospects(conn, intel_report_id, brand_id)
    conn.close()

    output_path.parent.mkdir(parents=True, exist_ok=True)
    with open(output_path, "w", newline="", encoding="utf-8-sig") as f:
        writer = csv.writer(f)
        writer.writerow(_HEADER_COLS)
        for row in rows:
            writer.writerow([row[col] for col in _DB_COLS])

    print(f"[export_prospects] 已导出 {len(rows)} 条记录到: {output_path}")
    return len(rows)


def main() -> None:
    parser = argparse.ArgumentParser(description="customer_prospect 批量导出工具")
    parser.add_argument("-o", "--output", required=True, help="输出文件路径（.xlsx 或 .csv）")
    parser.add_argument("--report_id", type=int, default=None, help="按 intel_report_id 过滤")
    parser.add_argument("--brand_id", default=None, help="按 brand_id 过滤")
    parser.add_argument(
        "--format", choices=["excel", "csv"], default="excel",
        help="输出格式（默认 excel）"
    )
    args = parser.parse_args()

    output_path = Path(args.output)
    # 根据扩展名自动判断格式
    if output_path.suffix.lower() == ".csv":
        fmt = "csv"
    elif output_path.suffix.lower() in (".xlsx", ".xls"):
        fmt = "excel"
    else:
        fmt = args.format

    if fmt == "csv":
        count = export_to_csv(output_path, args.report_id, args.brand_id)
    else:
        count = export_to_excel(output_path, args.report_id, args.brand_id)

    print(f"完成，共 {count} 条")


if __name__ == "__main__":
    main()
```
