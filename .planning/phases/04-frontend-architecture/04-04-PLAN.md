    ```typescript
    import { NextRequest, NextResponse } from 'next/server'
    import { createClient } from '@/lib/supabase/server'

    export async function GET(request: NextRequest) {
      const { searchParams } = new URL(request.url)
      const industry_l1 = searchParams.get('industry_l1')
      const industry_l2 = searchParams.get('industry_l2')
      const relation = searchParams.get('relation')
      const mds = searchParams.get('mds')

      const supabase = await createClient()

      let query = supabase
        .from('exhibition_brand')
        .select(`
          brand_id, name_cn, name_en, organizer, competition_relation, mds_related,
          industry_l1, industry_l2, city,
          exhibition_edition!inner(
            area_sqm, exhibitors_count, visitors_count,
            yoy_trend, year, date_start, date_end, edition_id
          )
        `)

      if (industry_l1) query = query.eq('industry_l1', industry_l1)
      if (industry_l2) query = query.eq('industry_l2', industry_l2)
      if (relation && relation !== '全部') query = query.eq('competition_relation', relation)
      if (mds && mds !== '全部') query = query.eq('mds_related', mds)

      const { data, error } = await query
      if (error) return NextResponse.json({ error: error.message }, { status: 500 })

      // KPI 聚合（取最新一届的数据）
      const kpi = {
        area_sqm: data.reduce((s, b) => s + ((b as any).exhibition_edition?.[0]?.area_sqm ?? 0), 0),
        exhibitors_count: data.reduce((s, b) => s + ((b as any).exhibition_edition?.[0]?.exhibitors_count ?? 0), 0),
        visitors_count: data.reduce((s, b) => s + ((b as any).exhibition_edition?.[0]?.visitors_count ?? 0), 0),
        organizer_count: new Set(data.map(b => (b as any).organizer).filter(Boolean)).size,
      }

      // 行业分布（industry_l2 分组计数）
      const industryDistribution: Record<string, number> = {}
      data.forEach(b => {
        const l2 = (b as any).industry_l2
        if (l2) industryDistribution[l2] = (industryDistribution[l2] || 0) + 1
      })

      return NextResponse.json({
        brands: data,
        kpi,
        industryDistribution: Object.entries(industryDistribution).map(([name, value]) => ({ name, value })),
      })
    }
    ```

    export async function GET() {
      const supabase = await createClient()

      // 角色验证
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

      const { data, error } = await supabase.auth.admin.listUsers()

      return NextResponse.json({
        users: data.users.map(u => ({
          id: u.id,
          email: u.email,
          role: u.app_metadata?.role || 'readonly',
          created_at: u.created_at,
          last_sign_in_at: u.last_sign_in_at,
        }))
